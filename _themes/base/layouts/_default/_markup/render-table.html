{{/* 마크다운 표 렌더 훅 (Hugo 0.134+). 셀·글의 표를 가로 스크롤 틀(.tw)로 감싸고 "+…" 칸은 up(빨강), "-…"·"−…" 칸은 dn(파랑)을 단다 — 한국식 등락 색. ⛔글자 하나짜리 "-"(빈칸 표시)는 색을 달지 않는다. */}}
{{- range .THead }}{{ range . }}{{ .Text }}{{ end }}{{ end }} {{- range .TBody }} {{- range . }} {{- $t := .Text | plainify | strings.TrimSpace }} {{- $cls := "" }} {{- if gt (strings.RuneCount $t) 1 }} {{- if hasPrefix $t "+" }}{{ $cls = "up" }}{{ else if or (hasPrefix $t "-") (hasPrefix $t "−") }}{{ $cls = "dn" }}{{ end }} {{- end }} {{ .Text }} {{- end }} {{ end }}
