{{/* 종목별 20거래일 차이의 분포 — 인라인 SVG. ⛔파일로 굽지 않는다. 셀마다 파일을 만들면 신호 25개에서 CF 배포당 파일 상한(20,000)을 넘는다(설계안 ⛔9). 인자 dict: sig(grid.json 의 신호 하나) · w · h · aria · small(축 글자 뺌) mark(이 종목 값 · 선택) · mlabel(표식 글자) ⛔mark 가 0.00 일 수 있다 — `with` 로 거르면 사라진다. isset 으로 본다. grid.hist 는 0 이 칸 경계에 오게 잘라 두었다 — 칸마다 위/아래 색이 하나로 정해진다. */}} {{- $s := .sig }}{{ $hs := $s.hist }} {{- $W := float (.w | default 720) }}{{ $H := float (.h | default 180) }} {{- $small := .small | default false }} {{- $pl := 10.0 }}{{ $pr := 10.0 }}{{ $pt := cond $small 8.0 24.0 }}{{ $pb := cond $small 6.0 22.0 }} {{- $k := len $hs.counts }} {{- $bw := div (sub (sub $W $pl) $pr) (float $k) }} {{- $ph := sub (sub $H $pt) $pb }} {{- $mx := float $hs.max }} {{- $lo := float $hs.lo }}{{ $st := float $hs.step }} {{- $hi := add $lo (mul $st (float $k)) }} {{- $base := add $pt $ph }} {{- $has := isset . "mark" }} {{- $mv := 0.0 }}{{ if $has }}{{ $mv = float .mark }}{{ end }}
